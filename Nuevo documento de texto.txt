drop database if exists carniceria;
create database carniceria;
use carniceria;

create table clientes(
IdCliente int primary key not null,
Nombre varchar(20),
Telefono int(10)
);

create table Productos(
IdProducto int primary key not null,
Nombre varchar(30),
Precio decimal
);

create table MetodosPago(
IdMetodoPago int primary key not null,
Nombre varchar(20)
);

create table Carniceros(
IdCarnicero int primary key not null,
Nombre varchar(20),
Apellido varchar(20),
Edad int
);

create table Pedidos(
IdPedido int primary key not null,
IdCarnicero int,
IdCliente int,
IdMetodoPago int,
TotalPrecio decimal,
foreign key Carniceros(IdCarnicero) references IdCarnicero,
foreign key Clientes(IdCliente) references IdClientes,
foreign key MetodosPago(IdMetodoPago) references IdMetodoPago
);

create table ProductosPedidos(
idProductoPedido int primary key not null,
idProducto int,
idPedido int,
foreign key Productos(idProducto) references idProducto,
foreign key Pedidos(idPedido) references idPedido
);

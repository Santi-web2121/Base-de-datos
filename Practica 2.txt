drop database if exists streaming;
create database streaming;
use streaming;

create table series(
IDSerie int primary key not null,
Nombreserie varchar(30),
Anioestreno year,
Creador varchar(30)
);

create table personajes(
IDPersonaje int primary key not null,
Nombrepersonaje varchar(20), 
Idserie int,
foreign key (Idserie) references series (Idserie),
Rol varchar(20),
Nivelenergia varchar(20)
);

create table episodios(
IDEpisodio int primary key not null,
Tituloepisodio varchar(30),
Idserie int,
foreign key (Idserie) references series (Idserie),
Duracionmins int,
Rating10 float
); 

create table habilidades(
IDHabilidad int primary key not null,
idpersonaje int,
foreign key (idpersonaje) references personajes (idpersonaje),
nombreha varchar(30),
tipoha varchar(20)
);

create table enemigos(
IDRivalidad int primary key not null,
IDHeroe int,
foreign key (idheroe) references personajes (idpersonaje),
IDVillano int,
foreign key (idvillano) references personajes (idpersonaje),
EncuentrosTotales int
);

create table objetosMisticos(
IDObjeto int primary key not null,
NombreObj varchar(30),
IDDuenio int,
foreign key (idduenio) references personajes (idpersonaje),
ValorSubasta int
);
